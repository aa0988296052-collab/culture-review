    // 2. 核心功能：渲染畫面 (Render Function)
    function render() {
        for (const [key, items] of Object.entries(data)) {
            const container = document.getElementById(key);
            // 防呆機制：如果找不到對應的區塊，就跳過
            if (!container) {
                console.warn(`警告：HTML 中找不到 id 為 ${key} 的區塊`);
                continue;
            }
            
            // 清空舊內容（避免重複渲染）
            container.innerHTML = '';

            items.forEach(item => {
                const card = document.createElement('div');
                card.className = 'card';
                card.onclick = function() {
                    this.classList.toggle('open');
                    const ans = this.querySelector('.answer');
                    // 簡單的切換顯示
                    ans.style.display = ans.style.display === 'block' ? 'none' : 'block';
                };
                
                card.innerHTML = `
                    <h3><span><span class="tag">${item.tag}</span>${item.title}</span></h3>
                    <div class="answer">${item.content}</div>
                `;
                container.appendChild(card);
            });
        }
    }

    // 3. 核心功能：切換分頁 (Navigation)
    window.showSection = function(id) {
        // 隱藏所有區塊
        document.querySelectorAll('.section').forEach(el => el.classList.remove('active'));
        document.querySelectorAll('.btn').forEach(el => el.classList.remove('active'));
        
        // 顯示目標區塊
        const target = document.getElementById(id);
        if (target) {
            target.classList.add('active');
        } else {
            console.error(`錯誤：找不到 id 為 ${id} 的分頁`);
            return; // 中止執行
        }
        
        // 更新按鈕狀態
        const btnMap = {
            'intro': 0, 'types': 1, 'theory': 2, 'dilemma': 3
        };
        const buttons = document.querySelectorAll('.btn');
        if (buttons[btnMap[id]]) {
            buttons[btnMap[id]].classList.add('active');
        }
    }

    // ==========================================
    // 4. 自動化測試系統 (Auto Diagnostic System)
    // ==========================================
    function runSystemCheck() {
        const consoleDiv = document.getElementById('status-console');
        let logs = [];
        let errors = 0;

        function log(msg, isError = false) {
            const icon = isError ? '❌' : '✅';
            logs.push(`<div>${icon} ${msg}</div>`);
            if (isError) errors++;
        }

        // 測試 1: 檢查資料庫是否為空
        if (typeof data !== 'undefined' && Object.keys(data).length > 0) {
            log(`資料庫讀取成功 (共 ${Object.keys(data).length} 章節)`);
        } else {
            log(`嚴重錯誤：資料庫是空的！`, true);
        }

        // 測試 2: 檢查 HTML 元素是否存在
        const requiredIds = ['intro', 'types', 'theory', 'dilemma'];
        let missingIds = [];
        requiredIds.forEach(id => {
            if (!document.getElementById(id)) missingIds.push(id);
        });

        if (missingIds.length === 0) {
            log(`HTML 結構完整`);
        } else {
            log(`HTML 缺少區塊: ${missingIds.join(', ')}`, true);
        }

        // 測試 3: 檢查渲染是否成功 (模擬)
        const introCards = document.querySelectorAll('#intro .card');
        if (introCards.length > 0) {
            log(`渲染系統運作正常 (Intro 區塊有 ${introCards.length} 張卡片)`);
        } else {
            log(`渲染失敗：Intro 區塊沒有卡片`, true);
        }

        // 顯示結果
        if (errors === 0) {
            consoleDiv.style.background = '#27ae60'; // 綠色
            consoleDiv.innerHTML = `<strong>系統正常運行中</strong><br>` + logs.join('');
            // 3秒後自動隱藏成功訊息，以免擋路
            setTimeout(() => {
                consoleDiv.style.display = 'none';
            }, 3000);
        } else {
            consoleDiv.style.background = '#c0392b'; // 紅色
            consoleDiv.innerHTML = `<strong>系統偵測到錯誤</strong><br>` + logs.join('');
        }
    }

    // 初始化程式
    render();
    
    // 執行測試
    runSystemCheck();
